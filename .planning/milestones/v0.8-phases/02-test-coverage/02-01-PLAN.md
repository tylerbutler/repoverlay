**Step 1: Attempt to write a SIGPIPE regression test**

Try adding this test to tests/cli.rs under the Phase 2 section header:

```rust
#[test]
#[cfg(unix)]
fn sigpipe_does_not_cause_panic_when_pipe_closes_early() {
    use std::process::{Command, Stdio};
    use std::io::Read;

    // Spawn repoverlay list (or any subcommand that produces output) piped to a reader
    // that closes immediately. On Linux, SIGPIPE should cause clean exit (0 or 141),
    // not a panic with a Rust backtrace.
    let mut child = Command::new(assert_cmd::cargo::cargo_bin("repoverlay"))
        .args(["--help"])  // --help always produces output and exits quickly
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .expect("failed to spawn repoverlay");

    // Drop stdout immediately to trigger SIGPIPE if the process tries to write more
    drop(child.stdout.take());

    let status = child.wait().expect("failed to wait on child");

    // With SIGPIPE default restored, exit should be clean (0 from --help,
    // or 141/SIGPIPE on platforms that report pipe death — NOT a panic exit code)
    let code = status.code().unwrap_or(0);
    assert_ne!(code, 101, "exit code 101 indicates a Rust panic — SIGPIPE not handled");
}
```
